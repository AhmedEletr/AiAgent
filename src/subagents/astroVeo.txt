function cleanText(str: string): string {
  if (!str || typeof str !== "string") return "";
  return str
    .replace(/[\u0000-\u001F\u007F-\u009F]/g, "")
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/'/g, "&#039;");
}

function getAudioDuration(filePath: string): Promise<number> {
  return new Promise((resolve, reject) => {
    ffmpeg.ffprobe(filePath, (err, metadata) => {
      if (err) return reject(err);
      const duration = metadata?.format?.duration;
      resolve(typeof duration === "number" ? duration : parseFloat(duration || "13.0"));
    });
  });
}

function padAudioWithSilence(inputPath: string, outputPath: string, padSeconds: number): Promise<void> {
  return new Promise((resolve, reject) => {
    ffmpeg(inputPath)
      .audioFilters(`apad=pad_dur=${Math.max(0.1, padSeconds).toFixed(3)}`)
      .outputOptions(["-c:a libmp3lame", "-b:a 192k"])
      .save(outputPath)
      .on("end", () => resolve())
      .on("error", reject);
  });
}

function applyTempo(inputPath: string, outputPath: string, tempo: number): Promise<void> {
  const filters: string[] = [];
  let t = tempo;
  while (t > 2.0) { filters.push("atempo=2.0"); t = t / 2.0; }
  while (t < 0.5) { filters.push("atempo=0.5"); t = t / 0.5; }
  filters.push(`atempo=${Math.max(0.5, Math.min(2.0, t)).toFixed(4)}`);

  return new Promise((resolve, reject) => {
    ffmpeg(inputPath)
      .audioFilters(filters.join(","))
      .outputOptions(["-c:a libmp3lame", "-b:a 192k"])
      .save(outputPath)
      .on("end", () => resolve())
      .on("error", reject);
  });
}

function concatenateAudioFiles(audioFilePaths: string[], outputPath: string, tempDir: string): Promise<void> {
  return new Promise((resolve, reject) => {
    const concatListPath = path.join(tempDir, "audio_concat_manifest.txt");
    const manifestContent = audioFilePaths
      .map((file) => `file '${path.resolve(file).replace(/\\/g, "/")}'`)
      .join("\n");

    fs.writeFileSync(concatListPath, manifestContent, "utf-8");

    ffmpeg()
      .input(concatListPath)
      .inputOptions(["-f concat", "-safe 0"])
      .outputOptions(["-c:a libmp3lame", "-b:a 192k"])
      .save(outputPath)
      .on("end", () => resolve())
      .on("error", reject);
  });
}

function multiplexVideoAndAudio(videoPath: string, audioPath: string, outputPath: string): Promise<void> {
  return new Promise((resolve, reject) => {
    ffmpeg()
      .input(videoPath)
      .input(audioPath)
      .outputOptions([
        "-c:v libx264",
        "-preset fast",
        "-r 30",
        "-crf 18",
        "-pix_fmt yuv420p",
        "-c:a aac",
        "-b:a 192k",
        "-shortest"
      ])
      .save(outputPath)
      .on("end", () => resolve())
      .on("error", reject);
  });
}

async function synthesizeEdgeSpeech(text: string, outputPath: string, voice: string): Promise<void> {
  const tts = new MsEdgeTTS();
  await tts.setMetadata(voice, OUTPUT_FORMAT.AUDIO_24KHZ_48KBITRATE_MONO_MP3);
  const streamRes = tts.toStream(text.replace(/[<>&'"]/g, " "));
  const audioStream = (streamRes as any)?.audioStream || streamRes;
  const writeStream = fs.createWriteStream(outputPath);

  audioStream.pipe(writeStream);
  return new Promise((resolve, reject) => {
    writeStream.on("finish", () => resolve());
    audioStream.on("error", reject);
    writeStream.on("error", reject);
  });
}

  } catch {
    return "https://images.unsplash.com/photo-1518770660439-4636190af475?w=800&auto=format&fit=crop&q=80";

function cleanUnicodeAndSymbols(str: any): string {
  if (!str || typeof str !== "string") return "";
  return str
    .replace(/[\u0000-\u001F\u007F-\u009F]/g, "")
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/'/g, "&#039;");
}

function getAudioDuration(filePath: string): Promise<number> {
  return new Promise((resolve, reject) => {
    ffmpeg.ffprobe(filePath, (err, metadata) => {
      if (err) return reject(err);
      const duration = metadata?.format?.duration;
      resolve(typeof duration === "number" ? duration : parseFloat(duration || "13.0"));
    });
  });
}

function padAudioWithSilence(inputPath: string, outputPath: string, padSeconds: number): Promise<void> {
  return new Promise((resolve, reject) => {
    ffmpeg(inputPath)
      .audioFilters(`apad=pad_dur=${Math.max(0.1, padSeconds).toFixed(3)}`)
      .outputOptions(["-c:a libmp3lame", "-b:a 192k"])
      .save(outputPath)
      .on("end", () => resolve())
      .on("error", reject);
  });
}

function applyTempo(inputPath: string, outputPath: string, tempo: number): Promise<void> {
  const filters: string[] = [];
  let t = tempo;
  while (t > 2.0) { filters.push("atempo=2.0"); t /= 2.0; }
  while (t < 0.5) { filters.push("atempo=0.5"); t /= 0.5; }
  filters.push(`atempo=${Math.max(0.5, Math.min(2.0, t)).toFixed(4)}`);

  return new Promise((resolve, reject) => {
    ffmpeg(inputPath)
      .audioFilters(filters.join(","))
      .outputOptions(["-c:a libmp3lame", "-b:a 192k"])
      .save(outputPath)
      .on("end", () => resolve())
      .on("error", reject);
  });
}

function concatenateAudioFiles(audioFilePaths: string[], outputPath: string, tempDir: string): Promise<void> {
  return new Promise((resolve, reject) => {
    const manifestPath = path.join(tempDir, "audio_concat_manifest.txt");
    const manifestContent = audioFilePaths.map((f) => `file '${path.resolve(f).replace(/\\/g, "/")}'`).join("\n");
    fs.writeFileSync(manifestPath, manifestContent, "utf-8");

    ffmpeg()
      .input(manifestPath)
      .inputOptions(["-f concat", "-safe 0"])
      .outputOptions(["-c:a libmp3lame", "-b:a 192k"])
      .save(outputPath)
      .on("end", () => resolve())
      .on("error", reject);
  });
}

function multiplexVideoAndAudio(videoPath: string, audioPath: string, outputPath: string): Promise<void> {
  return new Promise((resolve, reject) => {
    ffmpeg()
      .input(videoPath)
      .input(audioPath)
      .outputOptions([
        "-c:v libx264",
        "-preset fast",
        "-r 30",
        "-crf 18",
        "-pix_fmt yuv420p",
        "-c:a aac",
        "-b:a 192k",
        "-shortest"
      ])
      .save(outputPath)
      .on("end", () => resolve())
      .on("error", reject);
  });
}

async function synthesizeSpeech(text: string, outputPath: string, voice: string): Promise<void> {
  const tts = new MsEdgeTTS();
  await tts.setMetadata(voice, OUTPUT_FORMAT.AUDIO_24KHZ_48KBITRATE_MONO_MP3);
  const streamRes = tts.toStream(text.replace(/[<>&'"]/g, " "));
  const audioStream = (streamRes as any)?.audioStream || streamRes;
  const writeStream = fs.createWriteStream(outputPath);

  return new Promise((resolve, reject) => {
    audioStream.pipe(writeStream);
    writeStream.on("finish", () => resolve());
    writeStream.on("error", reject);
    audioStream.on("error", reject);
  });
}
